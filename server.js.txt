const express = require('express');
const cors = require('cors');
const twilio = require('twilio'); // Replace with your chosen SMS provider's SDK

const app = express();
app.use(cors()); // Allows your local HTML file to talk to this server
app.use(express.json());

// Your SMS Provider Credentials (keep these secret, use environment variables in production)
const accountSid = 'YOUR_TWILIO_ACCOUNT_SID';
const authToken = 'YOUR_TWILIO_AUTH_TOKEN';
const twilioPhone = 'YOUR_TWILIO_PHONE_NUMBER';
const client = twilio(accountSid, authToken);

app.post('/send-otp', async (req, res) => {
    const { mobile, otp } = req.body;

    try {
        // Trigger the real SMS
        await client.messages.create({
            body: `Your Login OTP for Harishchandrapur-I Dev. Block is: ${otp}. Do not share this with anyone.`,
            from: twilioPhone,
            to: `+91${mobile}` // Assuming Indian mobile numbers (+91)
        });

        res.json({ success: true, message: "SMS dispatched successfully" });
    } catch (error) {
        console.error("SMS Error:", error);
        res.status(500).json({ success: false, message: "Failed to send SMS" });
    }
});

app.listen(3000, () => {
    console.log('Backend server running on http://localhost:3000');
});
